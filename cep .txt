package ProjQualquer;

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;

public class ConsumirCep {

    public static void main(String[] args) {

        // Dados do aluno
        System.out.println("==============================");
        System.out.println("CONSULTA DE CEP");
        System.out.println("Nome: LuisGustavoCroner");
        System.out.println("RA: 0023948");
        System.out.println("==============================");

        String cep = "01001000";

        String url = "https://viacep.com.br/ws/" + cep + "/json/";

        try {

            HttpClient client = HttpClient.newHttpClient();

            HttpRequest request = HttpRequest.newBuilder()
                    .uri(URI.create(url))
                    .GET()
                    .build();

            HttpResponse<String> response = client.send(
                    request,
                    HttpResponse.BodyHandlers.ofString()
            );

            System.out.println("\nResultado da consulta:");
            System.out.println(response.body());

        } catch (Exception e) {

            System.out.println(
                    "Erro ao consultar o CEP: " + e.getMessage()
            );

        }
    }
}
